---
title: submitTxBlock
---

The `submitTxBlock` function signs, broadcasts, and waits for a transaction to be included in a block by skipping the InterwovenKit UI modal and going directly to the user's browser wallet confirmation, returning the complete transaction response. This provides custom transaction UX with immediate confirmation results.

<Info>
  This method skips the InterwovenKit fee selection modal and requires explicit fee specification. Use this when building custom transaction confirmation UI that needs immediate transaction results.
</Info>

## Function Signature

```tsx
function submitTxBlock(txParams: TxParams): Promise<DeliverTxResponse>
```

<ParamField path="txParams" type="TxParams" required>
  Transaction parameters including messages, explicit fee, and optional parameters.
</ParamField>

<ParamField path="Returns" type="Promise<DeliverTxResponse>">
  Promise that resolves to the complete transaction response including execution results, events, and gas usage.
</ParamField>

## Key Differences from requestTxBlock

| Feature | `submitTxBlock` | `requestTxBlock` |
|---------|----------------|------------------|
| **InterwovenKit Modal** | ❌ Skipped | ✅ Shows fee selection UI |
| **Fee Required** | ✅ Must provide `StdFee` | ❌ Optional (UI handles fee) |
| **User Confirmation** | Browser wallet | InterwovenKit UI |
| **Parameter Type** | `TxParams` | `TxRequest` |
| **Response** | `DeliverTxResponse` | `DeliverTxResponse` |

## Basic Usage

```tsx
import { calculateFee, GasPrice } from "@cosmjs/stargate"
import { useInterwovenKit } from "@initia/interwovenkit-react"
import type { TxParams, EncodeObject } from "@initia/interwovenkit-react"

function CustomTransactionWithResults() {
  const { submitTxBlock, estimateGas, address } = useInterwovenKit()

  const sendWithCustomFlowAndResults = async () => {
    const message: EncodeObject = {
      typeUrl: "/cosmos.bank.v1beta1.MsgSend",
      value: {
        fromAddress: address,
        toAddress: "init1recipient...",
        amount: [{ denom: "uinit", amount: "1000000" }]
      }
    }

    try {
      // Calculate fee programmatically
      const gasLimit = await estimateGas({ messages: [message] })
      const fee = calculateFee(gasLimit, GasPrice.fromString("0.015uinit"))

      const txParams: TxParams = {
        messages: [message],
        fee, // Required - explicit fee
        memo: "Direct transaction with block confirmation"
      }

      // Skip InterwovenKit modal, go to wallet and wait for block
      const response = await submitTxBlock(txParams)

      if (response.code === 0) {
        console.log("✅ Transaction successful!")
        console.log("Hash:", response.transactionHash)
        console.log("Block height:", response.height)
        console.log("Gas used:", response.gasUsed)
        console.log("Events:", response.events.length)
        return response
      } else {
        console.error("❌ Transaction failed:", response.rawLog)
        throw new Error(`Transaction failed: ${response.rawLog}`)
      }
    } catch (error) {
      console.error("Direct block transaction failed:", error)
      throw error
    }
  }

  return (
    <button onClick={sendTokensWithConfirmation}>
      Send Tokens (Custom Flow + Results)
    </button>
  )
}
```
